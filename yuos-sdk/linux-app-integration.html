Integrating an Existing Linux App with Yu OS

Wrap an existing C/C++ Linux binary or integrate your existing Qt, GTK, Electron, or native Linux application with the Yu OS SDK. Select your preferred SDK wrapper language in the top-right tab bar.

1. Wrap an Existing C/C++ Linux Binary

Keep your compiled C/C++ binary in bin/my-cpp-tool and wrap it with the SDK in your chosen language to add Yu OS AI and sandboxed permissions. Note that declaring "kind": "path" in app.json automatically grants read access to the user-supplied file without any needs boilerplate:

{
  "id": "my-analyzer",
  "version": "1.0.0",
  "name": {"en": "My Analyzer"},
  "ai": {},
  "operations": {
    "analyze": {
      "label": {"en": "Analyze a file and summarize results"},
      "args": [{"name": "file", "kind": "path", "required": true}],
      "needs": ["ai.chat"]
    }
  }
}

Wrapper in Python (main.py):

import subprocess
from yuos_sdk.ai import chat
from yuos_sdk.app import command, desktop

@command
def analyze(args: list[str]) -> dict:
    raw = subprocess.check_output(["./bin/my-cpp-tool", args[0]], text=True)
    summary = chat(raw, system="Summarize the output in one sentence.").output.text
    return {"summary": summary}

@desktop
def main() -> None:
    subprocess.run(["./bin/my-cpp-tool", "--gui"], check=True)

Wrapper in Node.js / TypeScript (index.ts):

import { execFileSync, spawnSync } from "node:child_process";
import { ai, app } from "@yuos/sdk";

app.command(async function analyze(args: string[]) {
  const raw = execFileSync("./bin/my-cpp-tool", [args[0]], { encoding: "utf8" });
  const reply = await ai.chat(raw, { system: "Summarize the output in one sentence." });
  return { summary: reply.output.text };
});

app.desktop(() => {
  spawnSync("./bin/my-cpp-tool", ["--gui"], { stdio: "inherit" });
});

app.run();

Wrapper in Go (main.go):

package main

import (
	"log"
	"os/exec"
	yuossdk "github.com/xiaoyu-work/Yu-OS/yuos-sdk/go"
)

func main() {
	yuossdk.HandleCommand("analyze", func(args []string) (any, error) {
		out, err := exec.Command("./bin/my-cpp-tool", args[0]).Output()
		if err != nil {
			return nil, err
		}
		reply, err := yuossdk.Chat(string(out), yuossdk.ChatOptions{System: "Summarize in one sentence."})
		if err != nil {
			return nil, err
		}
		return map[string]any{"summary": reply.Output.Text}, nil
	})
	yuossdk.HandleDesktop(func() error {
		return exec.Command("./bin/my-cpp-tool", "--gui").Run()
	})
	if err := yuossdk.RunApp(); err != nil {
		log.Fatal(err)
	}
}

Wrapper in Rust (src/main.rs):

use std::process::Command;
use yuos_sdk::{ai, app};

fn main() -> Result<(), app::AppError> {
    app::App::new()
        .command("analyze", |args| {
            let out = Command::new("./bin/my-cpp-tool").arg(&args[0]).output()?;
            let text = String::from_utf8_lossy(&out.stdout);
            let reply = ai::chat(&text, ai::ChatOpts::new().system("Summarize in one sentence."))?;
            Ok(serde_json::json!({ "summary": reply.output.text }))
        })
        .desktop(|| {
            Command::new("./bin/my-cpp-tool").arg("--gui").status()?;
            Ok(())
        })
        .run()
}

2. Adapt an Existing Desktop GUI App (File Picker, Data Dir & Notifications)

Replace hardcoded config paths with data_dir and use open_file so user-selected files are automatically granted read access inside the sandbox:

from pathlib import Path
from yuos_sdk import gui
from yuos_sdk.ai import chat_stream
from yuos_sdk.app import data_dir, desktop, notify

@desktop
def main() -> None:
    chosen = gui.open_file("Select a document")
    if not chosen:
        return
    text = Path(chosen).read_text(encoding="utf-8")
    chunks = list(chat_stream(text, system="Summarize key points."))
    out = data_dir() / "last_summary.txt"
    out.write_text("".join(chunks), encoding="utf-8")
    notify("Summary saved", str(out))
import * as fs from "node:fs";
import * as path from "node:path";
import { ai, app, gui } from "@yuos/sdk";

app.desktop(async () => {
  const chosen = gui.openFile("Select a document");
  if (!chosen) return;
  const text = fs.readFileSync(chosen, "utf8");
  let summary = "";
  for await (const chunk of ai.chatStream(text, { system: "Summarize key points." })) {
    summary += chunk;
  }
  const out = path.join(app.dataDir(), "last_summary.txt");
  fs.writeFileSync(out, summary, "utf8");
  app.notify("Summary saved", out);
});
yuossdk.HandleDesktop(func() error {
	chosen, err := yuossdk.OpenFile("Select a document")
	if err != nil || chosen == "" {
		return err
	}
	raw, err := os.ReadFile(chosen)
	if err != nil {
		return err
	}
	reply, err := yuossdk.Chat(string(raw), yuossdk.ChatOptions{System: "Summarize key points."})
	if err != nil {
		return err
	}
	dir, _ := yuossdk.DataDir()
	out := filepath.Join(dir, "last_summary.txt")
	_ = os.WriteFile(out, []byte(reply.Output.Text), 0o600)
	return yuossdk.Notify("Summary saved", out)
})
app::App::new().desktop(|| {
    if let Some(chosen) = yuos_sdk::gui::open_file("Select a document")? {
        let text = std::fs::read_to_string(chosen)?;
        let reply = yuos_sdk::ai::chat(&text, yuos_sdk::ai::ChatOpts::new().system("Summarize key points."))?;
        let out = yuos_sdk::app::data_dir()?.join("last_summary.txt");
        std::fs::write(&out, &reply.output.text)?;
        yuos_sdk::app::notify("Summary saved", &out.display().to_string())?;
    }
    Ok(())
})

3. Expose Tools to the Yu OS System Agent (MCP)

from yuos_sdk.mcp import App

app = App.from_manifest()

@app.tool("my-analyzer.analyze")
def analyze_tool(file: str) -> dict:
    return {"summary": f"Analyzed {file}"}

if __name__ == "__main__":
    app.serve()
import { mcp } from "@yuos/sdk";

const server = mcp.App.fromManifest();
server.tool("my-analyzer.analyze", async ({ file }) => ({ summary: `Analyzed ${file}` }));
server.serve();
mcpApp, err := yuossdk.LoadMCPApp()
if err != nil {
	log.Fatal(err)
}
mcpApp.Bind("my-analyzer.analyze", func(ctx *yuossdk.MCPCallContext, args map[string]any) (*yuossdk.ToolResult, error) {
	return yuossdk.StructuredResult(map[string]any{"summary": args["file"]})
})
_ = mcpApp.ServeStdio()
let mut mcp_app = yuos_sdk::mcp::App::from_environment()?;
// Bind tool implementations and run:
mcp_app.serve_stdio().await?;
