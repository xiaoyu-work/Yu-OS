Yu OS SDK Quickstart

Build, test, and package your first Yu OS App with command-line, desktop, and system AI support. Select your language in the top-right tab bar.

1. Install the SDK

Official repository & releases: https://github.com/xiaoyu-work/Yu-OS/releases/tag/yuos-sdk-v0.1.0.dev2026112401

Requires Python 3.10+. On Yu OS, yuos-sdk is preinstalled at /usr/lib/yuos/python/. Outside Yu OS, install directly from the GitHub Release:

python3 -m pip install https://github.com/xiaoyu-work/Yu-OS/releases/download/yuos-sdk-v0.1.0.dev2026112401/yuos_sdk-0.1.0.dev2026112401-py3-none-any.whl

Requires Node.js 18+. Install directly from the GitHub Release tarball:

npm install https://github.com/xiaoyu-work/Yu-OS/releases/download/yuos-sdk-v0.1.0.dev2026112401/yuos-sdk-node-0.1.0.dev2026112401.tgz

Requires Go 1.21+.

go get github.com/xiaoyu-work/Yu-OS/yuos-sdk/go@v0.1.0-dev.2026112401

Requires Rust (Edition 2021). Add to your Cargo.toml:

[dependencies]
yuos-sdk = { git = "https://github.com/xiaoyu-work/Yu-OS.git", tag = "yuos-sdk-v0.1.0.dev2026112401" }

3. Write CLI and Desktop Entrypoints

Save as hello-ai/main.py:

from yuos_sdk.ai import chat
from yuos_sdk.app import command, desktop

def summarize_text(text: str) -> str:
    return chat(text, system="Summarize in one short sentence.").output.text

@command
def summarize(args: list[str]) -> dict:
    return {"summary": summarize_text(args[0])}

@desktop
def main() -> None:
    print(summarize_text("Desktop mode launched."))

Save as hello-ai/index.ts:

import { ai, app } from "@yuos/sdk";

async function summarizeText(text: string): Promise<string> {
  return (await ai.chat(text, { system: "Summarize in one short sentence." })).output.text;
}

app.command(async function summarize(args: string[]) {
  return { summary: await summarizeText(args[0]) };
});

app.desktop(async () => {
  console.log(await summarizeText("Desktop mode launched."));
});

app.run();

Save as hello-ai/main.go and build to hello-ai/bin/hello-ai:

package main

import (
	"fmt"
	"log"
	yuossdk "github.com/xiaoyu-work/Yu-OS/yuos-sdk/go"
)

func summarizeText(text string) (string, error) {
	reply, err := yuossdk.Chat(text, yuossdk.ChatOptions{System: "Summarize in one short sentence."})
	if err != nil {
		return "", err
	}
	return reply.Output.Text, nil
}

func main() {
	yuossdk.HandleCommand("summarize", func(args []string) (any, error) {
		out, err := summarizeText(args[0])
		if err != nil {
			return nil, err
		}
		return map[string]any{"summary": out}, nil
	})
	yuossdk.HandleDesktop(func() error {
		out, err := summarizeText("Desktop mode launched.")
		if err == nil {
			fmt.Println(out)
		}
		return err
	})
	if err := yuossdk.RunApp(); err != nil {
		log.Fatal(err)
	}
}

Save as hello-ai/src/main.rs and build to hello-ai/bin/hello-ai:

use yuos_sdk::{ai, app};

fn summarize_text(text: &str) -> Result<String, ai::AiError> {
    Ok(ai::chat(text, ai::ChatOpts::new().system("Summarize in one short sentence."))?.output.text)
}

fn main() -> Result<(), app::AppError> {
    app::App::new()
        .command("summarize", |args| {
            Ok(serde_json::json!({ "summary": summarize_text(&args[0])? }))
        })
        .desktop(|| {
            println!("{}", summarize_text("Desktop mode launched.")?);
            Ok(())
        })
        .run()
}

4. Test Locally

YUOS_DEV_AI_URL=http://localhost:11434/v1 yu app dev ./hello-ai summarize "Yu OS lets Apps call the user's configured model."
YUOS_DEV_AI_URL=http://localhost:11434/v1 yu app dev ./hello-ai --gui

5. Package into .deb & Submit to the Store

yu app pack ./hello-ai
yu app submit yuos-app-hello-ai_0.1.0_all.deb
